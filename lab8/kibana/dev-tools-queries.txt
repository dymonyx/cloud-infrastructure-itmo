# Полнотекстовый поиск по message
GET chainwise-logs/_search
{
  "size": 20,
  "query": {
    "match": {
      "message": "цепи"
    }
  },
  "sort": [
    {"timestamp": "asc"}
  ]
}

# Фильтр ERROR и временной диапазон
GET chainwise-logs/_search
{
  "size": 20,
  "query": {
    "bool": {
      "filter": [
        {"term": {"level": "ERROR"}},
        {
          "range": {
            "timestamp": {
              "gte": "2026-11-22T10:00:00Z",
              "lte": "2026-11-22T10:59:59Z"
            }
          }
        }
      ]
    }
  },
  "sort": [
    {"timestamp": "asc"}
  ]
}

# Поиск одной операции по trace_id
GET chainwise-logs/_search
{

  "query": {
    "term": {
      "trace_id": "trace-ride-1007"
    }
  },
  "sort": [

  ]
}

# Агрегация количества логов по level
